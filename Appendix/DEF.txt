// Install-Package EntityFramework


using System.Data.Entity;

internal class Program
{
    static void Main(string[] args)
    {
        var db = new NorthwindContext();
        var londonCustomers = db.Customers.Where(c => c.City == "London").ToList();
        foreach (var c in londonCustomers)
            Console.WriteLine($"{c.CustomerID,-8}{c.CompanyName,-35}{c.City}");
    }
}

public class Customer
{
    public string CustomerID { get; set; }
    public string CompanyName { get; set; }
    public string Country { get; set; }
    public string City { get; set; }
}


public class NorthwindContext : DbContext
{
    public NorthwindContext()
        : base("Server=.;Database=Northwind;Trusted_Connection=True;")
    {
    }

    public DbSet<Customer> Customers { get; set; }
}
